cmake_minimum_required(VERSION 3.27)
project(qtpanel VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# GuiPrivate is requested explicitly: the composition path has no native window,
# so input is injected through QWindowSystemInterface, which lives in the QPA
# private headers. Gui itself arrives transitively via Quick, but the private
# target only exists when asked for by name.
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 QuickEffects Quick3D Network Xml Multimedia ShaderTools WebEngineQuick WebSockets GuiPrivate)
qt_standard_project_setup(REQUIRES 6.8)

add_subdirectory(qml)

option(QTPANEL_BUILD_TESTS "Build the QtTest suite" OFF)
if(QTPANEL_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests EXCLUDE_FROM_ALL)
endif()

qt_add_executable(qt-panel WIN32
    src/main.cpp
    src/core/HttpClient.h src/core/HttpClient.cpp
    src/core/Log.h src/core/Log.cpp
    src/core/QmlNetwork.h src/core/QmlNetwork.cpp

    src/core/SoundFx.h src/core/SoundFx.cpp
    src/core/SpeechService.h src/core/SpeechService.cpp
    src/core/TextFix.h src/core/TextFix.cpp
    src/services/camera/CameraClient.h src/services/camera/CameraClient.cpp

    src/services/news/NewsService.h src/services/news/NewsService.cpp
    src/services/pressreader/PressReaderService.h src/services/pressreader/PressReaderService.cpp

    src/services/starvis/StarvisService.h src/services/starvis/StarvisService.cpp

# C++/WinRT projections for the composition backdrop path. Generated by
# build.ps1 from the winmd committed under native/winappsdk-runtime, so a fresh
# clone needs no NuGet package and nothing installed on the machine.
set(WINAPPSDK_VENDOR ${CMAKE_CURRENT_SOURCE_DIR}/../winappsdk-runtime)
set(WINRT_GENERATED ${CMAKE_CURRENT_BINARY_DIR}/winrt-generated)

target_include_directories(qt-panel PRIVATE src ${WINRT_GENERATED} ${WINAPPSDK_VENDOR}/include)
target_compile_options(qt-panel PRIVATE /utf-8 /W4)
target_link_libraries(qt-panel PRIVATE
    Qt6::Quick
    # QPA private headers: the composition path has no native window, so input
    # has to be injected through QWindowSystemInterface. Nothing else uses this.
    Qt6::GuiPrivate
    Qt6::QuickControls2
    Qt6::QuickEffects
    Qt6::Quick3D
    Qt6::Network
    Qt6::Xml
    Qt6::Multimedia
    Qt6::WebSockets
    Qt6::WebEngineQuick
    qtpanel_uiplugin
    dwmapi
    user32
    bcrypt
    advapi32
    sapi          # offline speech (ISpVoice) for spoken alerts
    winmm         # native WAV fallback when Qt Multimedia cannot enumerate audio
    ole32
    windowsapp
    d3d11
    dxgi
)
